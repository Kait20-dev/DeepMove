cmake_minimum_required(VERSION 3.20)
project(DeepSpeedBot VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(DEEPSPEED_NATIVE "Tune for this CPU (binary may not be portable)" OFF)
option(DEEPSPEED_SANITIZE "Address/undefined sanitizers on GCC/Clang" OFF)
find_package(Threads REQUIRED)
add_library(deepspeed_core src/board.cpp src/evaluation.cpp src/search.cpp src/engine.cpp)
target_include_directories(deepspeed_core PUBLIC include)
target_link_libraries(deepspeed_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(deepspeed_core PRIVATE /W4 $<$<CONFIG:Release>:/O2>)
else()
  target_compile_options(deepspeed_core PRIVATE -Wall -Wextra -Wpedantic $<$<CONFIG:Release>:-O3>)
  if(DEEPSPEED_NATIVE)
    target_compile_options(deepspeed_core PRIVATE -march=native)
  endif()
  if(DEEPSPEED_SANITIZE)
    target_compile_options(deepspeed_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(deepspeed_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(deepspeed src/main.cpp)
target_link_libraries(deepspeed PRIVATE deepspeed_core)
if(MINGW)
  target_link_options(deepspeed PRIVATE -static)
endif()
enable_testing()
add_executable(deepspeed_tests tests/test_core.cpp)
target_link_libraries(deepspeed_tests PRIVATE deepspeed_core)
if(MINGW)
  target_link_options(deepspeed_tests PRIVATE -static)
endif()
add_test(NAME core COMMAND deepspeed_tests)
